cmake_minimum_required(VERSION 3.24)
if(CMAKE_BINARY_DIR STREQUAL CMAKE_SOURCE_DIR)
  message(FATAL_ERROR "Build Cane Godot outside the source tree.")
endif()
project(CaneGodot VERSION 0.1.0 LANGUAGES CXX)
set(CANE_GODOT_CPP "" CACHE PATH "Hash-pinned godot-cpp 4.5 source checkout")
if(NOT EXISTS "${CANE_GODOT_CPP}/gdextension/extension_api.json")
  message(FATAL_ERROR "Set CANE_GODOT_CPP to the documented godot-cpp checkout.")
endif()
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_DISABLE_EXCEPTIONS OFF CACHE BOOL "" FORCE)
set(GODOTCPP_USE_STATIC_CPP ON CACHE BOOL "" FORCE)
if(MSVC)
  # The distributed extension owns its C++ objects; Godot interaction uses its C ABI.
  # Set this before both subdirectories so Core, bindings and adapter use the same CRT.
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" CACHE STRING "" FORCE)
endif()
set(GODOTCPP_SYSTEM_HEADERS ON CACHE BOOL "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build-profile.json" CACHE PATH "" FORCE)
option(CANE_GODOT_REPRODUCIBLE_PATHS "Use portable source paths in distribution binaries" OFF)
if(CANE_GODOT_REPRODUCIBLE_PATHS)
  get_filename_component(CANE_CPP_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
  if(MSVC)
    # Distribution builds pin a compiler that supports deterministic path mapping.
    add_compile_options(/experimental:deterministic)
  endif()
  foreach(CANE_PATH_PAIR IN ITEMS
      "${CANE_CPP_SOURCE_ROOT}|cane-cpp"
      "${CANE_GODOT_CPP}|godot-cpp"
      "${CMAKE_BINARY_DIR}|build")
    string(REPLACE "|" ";" CANE_PATH_PARTS "${CANE_PATH_PAIR}")
    list(GET CANE_PATH_PARTS 0 CANE_PATH_FROM)
    list(GET CANE_PATH_PARTS 1 CANE_PATH_TO)
    if(MSVC)
      file(TO_NATIVE_PATH "${CANE_PATH_FROM}" CANE_PATH_NATIVE)
      add_compile_options("/pathmap:${CANE_PATH_NATIVE}=${CANE_PATH_TO}")
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      add_compile_options("-ffile-prefix-map=${CANE_PATH_FROM}=${CANE_PATH_TO}")
    else()
      message(FATAL_ERROR "Portable binary paths are not configured for this compiler.")
    endif()
  endforeach()
endif()
add_subdirectory("${CANE_GODOT_CPP}" godot-cpp)
set_target_properties(godot-cpp PROPERTIES PDB_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory(../cane-core cane-core)
add_library(cane-godot SHARED src/register_types.cpp src/skeleton_data.cpp src/skeleton.cpp src/playback.cpp src/host_controls.cpp src/canvas_projection.cpp src/texture_resources.cpp src/runtime_resources.cpp src/bone.cpp src/pose_followers.cpp src/follower.cpp src/point.cpp)
target_sources(cane-godot PRIVATE src/slot.cpp src/slot_nodes.cpp)
target_sources(cane-godot PRIVATE src/project_controls.cpp)
target_sources(cane-godot PRIVATE src/constraint_controls.cpp src/constraint_values.cpp)
target_sources(cane-godot PRIVATE src/canvas_batches.cpp)
target_sources(cane-godot PRIVATE src/serialized_data.cpp)
target_sources(cane-godot PRIVATE src/serialized_playback.cpp)
target_sources(cane-godot PRIVATE src/vertex_controls.cpp)
target_sources(cane-godot PRIVATE src/bounds.cpp)
target_sources(cane-godot PRIVATE src/pose_modifiers.cpp)
target_sources(cane-godot PRIVATE src/geometry_editor.cpp)
target_sources(cane-godot PRIVATE src/geometry_controls.cpp)
target_sources(cane-godot PRIVATE src/data_controls.cpp)
target_sources(cane-godot PRIVATE src/runtime_skin.cpp)
target_sources(cane-godot PRIVATE src/sdk_controls.cpp src/snapshot_values.cpp)
target_link_libraries(cane-godot PRIVATE Cane::Core godot::cpp)
target_compile_features(cane-godot PRIVATE cxx_std_17)
set_target_properties(cane-godot PROPERTIES CXX_EXTENSIONS OFF PREFIX "" OUTPUT_NAME "cane-godot")
if(MSVC)
  target_compile_options(cane-godot PRIVATE /W4 /WX /permissive- /fp:strict /utf-8)
  if(CANE_GODOT_REPRODUCIBLE_PATHS)
    target_link_options(cane-godot PRIVATE "/PDBALTPATH:%_PDB%")
  endif()
else()
  target_compile_options(cane-godot PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
endif()
